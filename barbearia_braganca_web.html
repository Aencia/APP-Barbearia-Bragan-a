📜 Script SQL para Criar Tabelas no Supabase

Cole o código abaixo no SQL Editor do seu painel Supabase para criar automaticamente as tabelas do sistema: 

-- SCRIPT DE CRIAÇÃO DAS TABELAS DA BARBEARIA BRAGANÇA
CREATE TABLE IF NOT EXISTS clients (
    id BIGINT PRIMARY KEY,
    name TEXT,
    phone TEXT,
    plan_name TEXT,
    plan_value NUMERIC,
    plan_active BOOLEAN,
    total_services INT
);

CREATE TABLE IF NOT EXISTS services (
    id BIGINT PRIMARY KEY,
    name TEXT,
    price NUMERIC,
    duration INT
);

CREATE TABLE IF NOT EXISTS products (
    id BIGINT PRIMARY KEY,
    name TEXT,
    stock INT,
    cost NUMERIC,
    price NUMERIC
);

CREATE TABLE IF NOT EXISTS barbers (
    id BIGINT PRIMARY KEY,
    name TEXT,
    phone TEXT,
    commission INT,
    is_active BOOLEAN
);

CREATE TABLE IF NOT EXISTS appointments (
    id BIGINT PRIMARY KEY,
    client_id BIGINT,
    client_name TEXT,
    barber_id BIGINT,
    barber_name TEXT,
    service_id BIGINT,
    service_name TEXT,
    date TEXT,
    time TEXT,
    payment_method TEXT,
    total_value NUMERIC,
    status TEXT
);

CREATE TABLE IF NOT EXISTS expenses (
    id BIGINT PRIMARY KEY,
    description TEXT,
    category TEXT,
    amount NUMERIC,
    date TEXT
);

-- DESABILITAR RLS PARA PERMITIR ACESSO DIRETO VIA CLIENT
